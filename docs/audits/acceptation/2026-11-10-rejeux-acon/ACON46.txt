== SOURCE C-LIVE-ACON46.py (sonde C-LIVE écrite pour ce rejeu)
"""C-LIVE-ACON46 — rejeu HTTP réel, DEUX connexions concurrentes (runserver multi-fils, Postgres) : Given C (lead L,
12 × 550 W) sans devis ; When deux POST generer-devis lâchés ensemble (barrière) ; Then 1 devis sur L, C lié à lui,
la seconde 200 deduplique: true. Témoin : deux clics successifs = 201 puis 200 deduplique. Comptes lus en base."""
import json
import subprocess
import sys
import threading

import requests

BASE = 'http://127.0.0.1:8028/api/django'
USER, PW = sys.argv[1], sys.argv[2]
FX = json.loads(sys.argv[3])


def session():
    # Jeton émis par Django (le login /token/ est limité en débit : 429 après quelques essais).
    prog = ("from django.contrib.auth import get_user_model;from rest_framework_simplejwt.tokens import AccessToken;"
            "print('@@'+str(AccessToken.for_user(get_user_model().objects.get(username=%r))))" % USER)
    out = subprocess.run(['docker', 'exec', 'accept-acon-dj', 'python', 'manage.py', 'shell', '-c', prog],
                         capture_output=True, text=True).stdout
    s = requests.Session()
    s.headers['Authorization'] = 'Bearer ' + next(x for x in out.splitlines() if x.startswith('@@'))[2:]
    return s


def compte(lead, cal):
    prog = ("from apps.ventes.models import Devis;from apps.calepinage.models import Calepinage;import json;"
            "d=list(Devis.objects.filter(lead_id=%d).values_list('pk','reference','statut'));"
            "print('@@'+json.dumps({'devis_du_lead':d,'calepinage_devis_id':Calepinage.objects.get(pk=%d).devis_id}))"
            % (lead, cal))
    out = subprocess.run(['docker', 'exec', 'accept-acon-dj', 'python', 'manage.py', 'shell', '-c', prog],
                         capture_output=True, text=True).stdout
    return json.loads(next(x for x in out.splitlines() if x.startswith('@@'))[2:])


res = {'concurrent': [], 'temoin': None}
for fx in FX[:-1]:
    s1, s2 = session(), session()
    barriere = threading.Barrier(2)
    out = [None, None]

    def go(i, s):
        barriere.wait()
        r = s.post(BASE + '/calepinage/calepinages/%d/generer-devis/' % fx['calepinage'], json={})
        try:
            j = r.json()
        except ValueError:
            j = {}
        out[i] = [r.status_code, j.get('devis'), j.get('deduplique'), round(r.elapsed.total_seconds(), 3)]

    t = [threading.Thread(target=go, args=(i, s)) for i, s in enumerate((s1, s2))]
    [x.start() for x in t]
    [x.join() for x in t]
    c = compte(fx['lead'], fx['calepinage'])
    res['concurrent'].append({'calepinage': fx['calepinage'], 'reponses': out, **c,
                              'un_seul_devis': len(c['devis_du_lead']) == 1,
                              'lie_au_devis': bool(c['devis_du_lead']) and c['calepinage_devis_id'] == c['devis_du_lead'][0][0]})
fx = FX[-1]
s = session()
r1 = s.post(BASE + '/calepinage/calepinages/%d/generer-devis/' % fx['calepinage'], json={})
r2 = s.post(BASE + '/calepinage/calepinages/%d/generer-devis/' % fx['calepinage'], json={})
res['temoin'] = {'http': [r1.status_code, r2.status_code], 'deduplique_2e': r2.json().get('deduplique'),
                 **compte(fx['lead'], fx['calepinage'])}
ok = all(x['un_seul_devis'] and x['lie_au_devis'] and sorted(y[0] for y in x['reponses']) == [200, 201]
         and any(y[2] is True for y in x['reponses']) for x in res['concurrent'])
res['repro'] = not ok
res['attendu_observe'] = ok
print(json.dumps(res, ensure_ascii=False, indent=1))
