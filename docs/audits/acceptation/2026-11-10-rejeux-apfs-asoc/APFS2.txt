# C-LIVE-APFS2 — rejeu HTTP réel : Django accept-apfs-dj (:8018) émet les jetons, FastAPI accept-apfs-fa (:8019,
# code du commit joué) reçoit les appels. Comptes de test de la base jetable uniquement.
import io
import json
import urllib.request
import urllib.error

DJ = 'http://127.0.0.1:8018'
FA = 'http://127.0.0.1:8019'
PWD = 'AccApfs-2026-tq7Wz'


def post(url, data=None, token=None, files=None):
    headers = {}
    body = None
    if files:
        b = '----accapfs'
        buf = io.BytesIO()
        for name, (fn, content, ct) in files.items():
            buf.write(('--%s\r\nContent-Disposition: form-data; name="%s"; filename="%s"\r\nContent-Type: %s\r\n\r\n'
                       % (b, name, fn, ct)).encode())
            buf.write(content + b'\r\n')
        buf.write(('--%s--\r\n' % b).encode())
        body = buf.getvalue()
        headers['Content-Type'] = 'multipart/form-data; boundary=' + b
    elif data is not None:
        body = json.dumps(data).encode()
        headers['Content-Type'] = 'application/json'
    if token:
        headers['Authorization'] = 'Bearer ' + token
    req = urllib.request.Request(url, data=body, headers=headers, method='POST')
    try:
        with urllib.request.urlopen(req, timeout=60) as r:
            return r.status, r.read()[:300].decode('utf-8', 'replace')
    except urllib.error.HTTPError as e:
        return e.code, e.read()[:300].decode('utf-8', 'replace')


def jeton(user):
    # le jeton d'accès est posé en cookie HttpOnly `access_token` (corps `{}`)
    req = urllib.request.Request(DJ + '/api/django/token/', method='POST',
                                 data=json.dumps({'username': user, 'password': PWD}).encode(),
                                 headers={'Content-Type': 'application/json'})
    with urllib.request.urlopen(req, timeout=30) as r:
        for h in r.headers.get_all('Set-Cookie') or []:
            h = h.strip()
            if h.startswith('access_token='):
                return h.split(';')[0].split('=', 1)[1]
    return None


out = {}
for nom, user in (('portail', 'acc_portail@example.invalid'), ('interne_directeur', 'acc_apfs_dir')):
    t = jeton(user)
    out[nom] = {'jeton': bool(t),
                'sql_agent_query': post(FA + '/sql-agent/query', {'question': 'liste des clients'}, t),
                'sql_agent_transcribe': post(FA + '/sql-agent/transcribe', token=t,
                                             files={'file': ('a.webm', b'\x1a\x45\xdf\xa3', 'audio/webm')}),
                'ocr_process_document': post(FA + '/ocr/process_document', token=t,
                                             files={'file': ('f.png', b'\x89PNG\r\n\x1a\n', 'image/png')})}
p = out['portail']
ok = all(p[k][0] == 403 and 'comptes internes' in p[k][1] for k in ('sql_agent_query', 'sql_agent_transcribe',
                                                                     'ocr_process_document'))
i = out['interne_directeur']
out['interne_passe_la_garde'] = all(i[k][0] != 403 for k in ('sql_agent_query', 'ocr_process_document'))
out['repro'] = ok and out['interne_passe_la_garde']
print(json.dumps(out, ensure_ascii=False, indent=1))
