def _base(slug):
    from decimal import Decimal
    from django.contrib.auth import get_user_model
    from rest_framework.test import APIClient
    from authentication.models import Company
    from apps.crm.models import Client
    co = Company.objects.create(nom='Live ' + slug, slug='live-' + slug.lower())
    u = get_user_model().objects.create_user(username='live_' + slug.lower(), password='x', company=co,
                                             role_legacy='admin')
    api = APIClient()
    api.force_authenticate(u)
    cl = Client.objects.create(company=co, nom='Client', prenom=slug, email=slug.lower() + '@example.invalid')
    return co, cl, api


def _systeme(co, cl, ref, annuel, mes, kwc='5.00'):
    from decimal import Decimal
    from apps.installations.models import Installation
    from apps.monitoring.models import MonitoringConfig
    inst = Installation.objects.create(company=co, reference=ref, client=cl,
                                       puissance_installee_kwc=Decimal(kwc), date_mise_en_service=mes)
    cfg = None
    if annuel:
        cfg = MonitoringConfig.objects.create(company=co, installation=inst, expected_annual_kwh=Decimal(annuel))
    return inst, cfg


def _profil(annuel, d1, d2, facteur=1):
    from decimal import Decimal
    from apps.monitoring.services import attendu_entre
    return (attendu_entre(Decimal(annuel), d1, d2) * Decimal(str(facteur))).quantize(Decimal('0.01'))


def _mensuels(co, inst, annuel, mois, facteur=1):
    import calendar
    from datetime import date, timedelta
    from apps.monitoring.models import ProductionReading
    for m in mois:
        jour = date(2026, m, 1)
        p = calendar.monthrange(2026, m)[1]
        ProductionReading.objects.create(company=co, installation=inst, date=jour, period_days=p,
                                         energy_kwh=_profil(annuel, jour, jour + timedelta(days=p - 1), facteur))


def _journaliers(co, inst, annuel, d1, d2, facteur=1):
    from datetime import timedelta
    from apps.monitoring.models import ProductionReading
    j = d1
    while j <= d2:
        ProductionReading.objects.create(company=co, installation=inst, date=j, period_days=1,
                                         energy_kwh=_profil(annuel, j, j, facteur))
        j += timedelta(days=1)


def _f(x):
    return None if x is None else float(x)

def sonde(ctx):
    import calendar, csv, io
    from datetime import timedelta
    from django.utils import timezone
    from apps.monitoring import analytics
    co, cl, api = _base('AMON100')
    today = timezone.localdate()
    inst, cfg = _systeme(co, cl, 'CHT-LAMON100', '12000', None, kwc='7.00')
    _journaliers(co, inst, '12000', today - timedelta(days=364), today)
    url = f'/api/django/monitoring/configs/{cfg.pk}/history/?months=12'
    r = api.get(url, HTTP_HOST='localhost')
    rows = r.data['data'] if r.status_code == 200 else []
    pr = {m['month']: _f(m['pr_pct']) for m in analytics.om_metrics(inst, today=today)['monthly_pr']}
    ecarts = {row['month']: (row['ratio_pct'], pr.get(row['month'])) for row in rows}
    egal = bool(rows) and [x['month'] for x in rows] == list(pr) and all(
        b is not None and a is not None and abs(a - b) <= 0.1 for a, b in ecarts.values())
    par_jour = [x['expected_kwh'] / calendar.monthrange(*(int(y) for y in x['month'].split('-')))[1]
                for x in rows[1:-1]]
    profil = bool(par_jour) and max(par_jour) / min(par_jour) > 1.5
    c = api.get(url + '&export=csv', HTTP_HOST='localhost')
    lignes = list(csv.DictReader(io.StringIO(b''.join(c.streaming_content).decode()))) if c.status_code == 200 else []
    csv_egal = [float(x['ratio_pct']) for x in lignes] == [x['ratio_pct'] for x in rows]
    cles = sorted(rows[0]) if rows else None
    return {'http': r.status_code, 'csv_http': c.status_code, 'cles': cles, 'ratio_vs_pr': ecarts,
            'attendu_jour_max_sur_min': round(max(par_jour) / min(par_jour), 3) if par_jour else None,
            'repro': egal and profil and csv_egal and cles == ['actual_kwh', 'expected_kwh', 'month', 'ratio_pct']}
