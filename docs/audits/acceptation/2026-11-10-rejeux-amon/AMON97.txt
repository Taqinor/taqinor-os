def _systeme(co, cl, ref, annuel, mes, kwc='5.00'):
    from decimal import Decimal
    from apps.installations.models import Installation
    from apps.monitoring.models import MonitoringConfig
    inst = Installation.objects.create(company=co, reference=ref, client=cl,
                                       puissance_installee_kwc=Decimal(kwc), date_mise_en_service=mes)
    cfg = None
    if annuel:
        cfg = MonitoringConfig.objects.create(company=co, installation=inst, expected_annual_kwh=Decimal(annuel))
    return inst, cfg


def _profil(annuel, d1, d2, facteur=1):
    from decimal import Decimal
    from apps.monitoring.services import attendu_entre
    return (attendu_entre(Decimal(annuel), d1, d2) * Decimal(str(facteur))).quantize(Decimal('0.01'))


def _mensuels(co, inst, annuel, mois, facteur=1):
    import calendar
    from datetime import date, timedelta
    from apps.monitoring.models import ProductionReading
    for m in mois:
        jour = date(2026, m, 1)
        p = calendar.monthrange(2026, m)[1]
        ProductionReading.objects.create(company=co, installation=inst, date=jour, period_days=p,
                                         energy_kwh=_profil(annuel, jour, jour + timedelta(days=p - 1), facteur))


def _journaliers(co, inst, annuel, d1, d2, facteur=1):
    from datetime import timedelta
    from apps.monitoring.models import ProductionReading
    j = d1

                                         energy_kwh=_profil(annuel, j, j, facteur))
        j += timedelta(days=1)
