def scenario(nom, produit, req1, req2):
    dans = threading.Event()
    liberer = threading.Event()
    res = {}
    n = {'n': 0}

    def crochet(*a, **k):
        if threading.current_thread().name == 'r1':
            n['n'] += 1
            if n['n'] == 1:
                dans.set()
                liberer.wait(30)
        return orig(*a, **k)

    def run(nm, f):
        try:
            r = f()
            res[nm] = {'http': r.status_code,
                       'detail': None if r.status_code < 400 else str(getattr(r, 'data', ''))[:160]}
        finally:
            connection.close()

    with mock.patch.object(stock_services, 'record_stock_movement', side_effect=crochet):
        t1 = threading.Thread(target=run, args=('r1', req1), name='r1')
        t1.start()
        assert dans.wait(60), 'r1 jamais entrée dans record_stock_movement'
        t2 = threading.Thread(target=run, args=('r2', req2), name='r2')
        t2.start()
        time.sleep(4)
        bloquee = t2.is_alive() and 'r2' not in res
        liberer.set()
        t1.join(60)
        t2.join(60)
    produit.refresh_from_db()
    mv = list(MouvementStock.objects.filter(produit=produit).order_by('id')
              .values_list('type_mouvement', 'quantite', 'quantite_avant', 'quantite_apres'))
    return {'r1': res.get('r1'), 'r2': res.get('r2'), 'r2_bloquee_pendant_r1': bloquee,
            'stock_final': produit.quantite_stock, 'mouvements': [tuple(str(x) for x in m) for m in mv]}
