function surveiller(p) {
  p.on('console', (m) => { if (m.type() === 'error') incidents['3'].push(`console.error ${m.text().slice(0, 200)}`) })
  p.on('pageerror', (e) => incidents['3'].push(`pageerror ${String(e).slice(0, 200)}`))
  p.on('dialog', async (d) => { incidents['4'].push(`dialog ${d.type()} ${d.message().slice(0, 120)}`); await d.dismiss().catch(() => {}) })
  p.on('request', (r) => {
    if (['PATCH', 'POST', 'PUT'].includes(r.method()) && r.url().includes(API)) {
      requetes.push({ m: r.method(), url: r.url().replace(FRONT, ''), body: r.postData() })
    }
  })
  p.on('response', (r) => {
    const u = r.url()
    if (!u.startsWith(FRONT || BASE)) return
    if (attenduUI.some((a) => a.re.test(u) && a.st === r.status())) return
    if (r.status() >= 500) incidents['1'].push(`${r.status()} ${u}`)
    else if (r.status() >= 400) incidents['2'].push(`${r.status()} ${u}`)
  })
}
const attenduUI = [] // refus ATTENDUS déclarés avant le geste (seuls excusés)

async function call(method, p, data, attendu, extra = {}) {
  const opts = { timeout: 180000, ...extra }
  if (data !== undefined) opts.data = data
  const r = await req[method.toLowerCase()](`${API}${p}`, opts)
  const st = r.status()
  let body = null
  const ct = r.headers()['content-type'] || ''
  const buf = await r.body()
  if (ct.includes('json')) { try { body = JSON.parse(buf.toString('utf8')) } catch { body = buf.toString('utf8') } } else body = buf
  const okAttendu = attendu === undefined ? st < 400 : (Array.isArray(attendu) ? attendu.includes(st) : st === attendu)
  if (!okAttendu && st >= 500) incidents['1'].push(`API ${method} ${p} -> ${st}`)
  else if (!okAttendu && st >= 400) incidents['2'].push(`API ${method} ${p} -> ${st} ${typeof body === 'object' && !Buffer.isBuffer(body) ? JSON.stringify(body).slice(0, 200) : ''}`)
  log(`${method} ${p} ${data !== undefined ? JSON.stringify(data).slice(0, 200) : ''} -> ${st}${Buffer.isBuffer(body) ? ` [${body.length} o, ${ct}]` : ` ${JSON.stringify(body).slice(0, 300)}`}`)
  return { st, body, ct }
}
const must = async (method, p, data, attendu) => {
  const r = await call(method, p, data, attendu)
  const ok = attendu === undefined ? r.st < 400 : (Array.isArray(attendu) ? attendu.includes(r.st) : r.st === attendu)
  if (!ok) throw new Error(`${method} ${p} -> ${r.st} ${JSON.stringify(r.body).slice(0, 400)}`)
  return r.body
}

}

// ── ATOT45 — prix négocié 9 000 sur 5 × 20 % émis : avoirs sous le créditable ──
async function etapeAvoir() {

  const p = await produit('Panneau Avoir', '1000')
  const cinq = Array.from({ length: 5 }, (_, i) => ({ libelle: `Tranche ${i + 1}`, unite: 'pct', pct_or_montant: 20 }))

    await must('PATCH', `/ventes/devis/${d.id}/`, { echeancier: cinq })
    d = await envoyerAccepter(d)
    const fids = []
    for (let i = 0; i < 5; i++) {
      const r = await call('POST', `/ventes/devis/${d.id}/generer-facture/`, {}, [200, 201])
      const fid = r.body && r.body.id
      if (fid) { if (r.body.statut === 'brouillon') await must('POST', `/ventes/factures/${fid}/emettre/`, {}); fids.push(fid) }
    }
    if (gesteAvant) await gesteAvant(fids)

    const avoirsAvant = liste(await must('GET', `/ventes/avoirs/?page_size=200`)).filter((a) => fids.includes(a.facture))
    return { d, fids, avant, avoirsAvant }
  }
  const res = {}
  // Cas 1 — 201, deux avoirs 2 400 + 600 sur deux factures distinctes, Σ dû 9 000
  const s1 = await scenario('cas1')
  verif('AVOIR', '5 tranches émises de 2 400 TTC (12 000)', s1.avant.length === 5 && s1.avant.every((f) => eq2(f.total_ttc, 2400) && f.statut !== 'brouillon'), s1.avant.map((f) => [f.reference, f.total_ttc, f.statut]))

  res.cas1 = { devis: s1.d.id, prix_negocie: { st: r1.st, body: r1.body }, factures: apres.map((f) => ({ id: f.id, ref: f.reference, ttc: f.total_ttc, avoirs_total: f.avoirs_total, du: f.montant_du, statut: f.statut })), avoirs: avoirs.map((a) => ({ id: a.id, ref: a.reference, facture: a.facture, ttc: a.total_ttc ?? a.montant_ttc, type: a.type_avoir ?? a.type })), somme_du: du }
  const montants = avoirs.map((a) => num(a.total_ttc ?? a.montant_ttc)).sort((x, y) => y - x)
  verif('AVOIR', 'prix-negocie 9 000 appliqué : 201', r1.st === 201, { st: r1.st })
  verif('AVOIR', 'deux avoirs 2 400 + 600 sur deux factures distinctes, chacun ≤ créditable', avoirs.length === 2 && eq2(montants[0], 2400) && eq2(montants[1], 600) && new Set(avoirs.map((a) => a.facture)).size === 2 && apres.every((f) => num(f.avoirs_total || 0) <= num(f.total_ttc) + 0.005), res.cas1.avoirs)

  if (page) { const c = await capture('ATOT-AVOIR-factures', `${FRONT}/ventes/factures`); res.capture = c.f }
  return res
}

// ── AGNR53 — échéancier recopié recalé au TTC de la copie ──────────────────────
async function etapeCopie() {
  const cl = await client('Copie')
  const p = await produit('Panneau Copie', '1000')
  const source = async (e) => {
    let d = await devis(cl, [{ p, q: '50', pu: '1000.00' }])

    d = await must('GET', `/ventes/devis/${d.id}/`)
    return { d, patch: pr.st }
  }
  const E1 = [{ libelle: 'Acompte', type: 'acompte', unite: 'montant', pct_or_montant: 6000 }, { libelle: 'Livraison', type: 'materiel', unite: 'pct', pct_or_montant: 60 }, { libelle: 'Solde', type: 'solde', unite: 'pct', pct_or_montant: 30 }]

  const somme = (e, ttc) => e.reduce((s, t) => s + (t.unite === 'montant' ? num(t.pct_or_montant) : num(ttc) * num(t.pct_or_montant) / 100), 0)

      await page.locator('input[type="password"]').fill(process.env.MDP || 'Demo@2026!')
      await page.getByRole('button', { name: 'Se connecter →' }).click()
      await page.waitForURL(/\/apps|\/dashboard|\/onboarding/, { timeout: 60000 })
      req = page.request
    } else {
      const anon = await pwRequest.newContext({ baseURL: BASE })
      const t = await anon.post(`${API}/token/`, { data: { username: process.env.LOGIN || 'demo_admin', password: process.env.MDP || 'Demo@2026!' } })
      const tb = await t.json().catch(() => ({}))
      log('token', t.status())
      const headers = tb.access ? { Authorization: `Bearer ${tb.access}` } : {}
      req = await pwRequest.newContext({ baseURL: BASE, extraHTTPHeaders: headers, storageState: await anon.storageState() })
    }
    const quoi = (process.env.ETAPES || 'bi,bc,avoir,copie,acompte,aller,releve,taux').split(',')
    const F = { bi: etapeBI, bc: etapeBC, avoir: etapeAvoir, copie: etapeCopie, acompte: etapeAcompte, aller: etapeAllerRetour, releve: etapeReleve, taux: etapeTaux }
    for (const q of quoi) {
      const inc0 = JSON.stringify(incidents)
      try { res[q] = await F[q]() } catch (e) { log(`ETAPE ${q} ERREUR`, String(e.stack || e).slice(0, 800)); verif(q.toUpperCase(), `étape ${q} exécutée sans erreur`, false, String(e).slice(0, 500)); if (page) await capture(`ERREUR-${q}`).catch(() => {}) }
      res[`incidents_${q}`] = JSON.parse(JSON.stringify(incidents))
      log(`INCIDENTS après ${q}`, inc0 === JSON.stringify(incidents) ? 'aucun nouveau' : incidents)
    }
  } finally {
    res.etapes = etapes
    res.incidents = incidents
    res.requetes = requetes.slice(-200)
    fs.writeFileSync(path.join(OUT, `rejeu-${mode}${process.env.SUFFIXE || ''}.json`), JSON.stringify(res, null, 2))
    fs.writeFileSync(path.join(OUT, `rejeu-${mode}${process.env.SUFFIXE || ''}.log`), journal.join('\n') + '\n')
    log('INCIDENTS', incidents)
    if (browser) await browser.close()
  }
})()
