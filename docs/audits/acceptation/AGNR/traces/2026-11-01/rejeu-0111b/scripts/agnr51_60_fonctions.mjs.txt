// Rejeu 0111b — AGNR51 / AGNR60 : les fonctions RÉELLES de echeancierEdition.js d'un arbre donné, appelées
// avec les Given des tâches (contre-épreuve de la moitié écran). Usage : node agnr51_60_fonctions.mjs <repo>
const repo = process.argv[2]
const m = await import(`${repo}/frontend/src/features/ventes/echeancierEdition.js`)
const eff = { industriel: [
  { libelle: 'Commande', jalon: 'commande', pct: 30 }, { libelle: 'Livraison', jalon: 'livraison_materiel', pct: 40 },
  { libelle: 'MES', jalon: 'mise_en_service', pct: 20 }, { libelle: 'Réception', jalon: 'reception_definitive', pct: 10 }] }
const out = {}
try { out.acompte_6000 = m.echeancierAvecAcompte(null, '6000', 17000, 'industriel', eff) } catch (e) { out.acompte_6000 = 'ERREUR ' + e }
try { out.acompte_8500 = m.echeancierAvecAcompte(null, '8500', 17000, 'industriel', eff) } catch (e) { out.acompte_8500 = 'ERREUR ' + e }
const servi = [
  { type: 'commande', libelle: 'Commande', unite: 'pct', pct_or_montant: 30, jalon: 'commande', fige_le: '2026-10-02T09:00:00+00:00' },
  { libelle: 'Livraison', unite: 'pct', pct_or_montant: 60, jalon: 'livraison_materiel', x_inconnue: 1 },
  { type: 'solde', libelle: 'Solde', unite: 'pct', pct_or_montant: 10 }]
const rt = m.saisieVersEcheancier(m.echeancierVersSaisie(servi))
out.aller_retour = rt
const canon = (v) => JSON.stringify(v, (k, x) => (x && typeof x === 'object' && !Array.isArray(x) ? Object.fromEntries(Object.keys(x).sort().map((c) => [c, x[c]])) : x))
out.aller_retour_identique = canon(rt) === canon(servi)
console.log(JSON.stringify(out, null, 1))
